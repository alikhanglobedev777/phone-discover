"""Import ONLY checked exact-model GSMArena product pictures from a bundled manifest.

The Phone Discover owner has permission to use GSMArena product imagery.
No URLs are invented at deployment. Gallery manifest was generated by
checking model-specific file names and actual GSMArena JPEG responses.

This updates primary pictures for matched Pixels, including the operator's
explicit requests for Pixel 7, 8a, 6a and 10 Pro. Previously genuine photos
remain nonprimary, and no phones, variants, offers, prices or colors change.
"""
import csv
import json
import re
from pathlib import Path
from urllib.parse import urlsplit

from django.db import migrations

PAGES = {
    "pixel 5a 5g": ("google_pixel_5a_5g-pictures-11059.php", "pixel-5a-5g"),
    "pixel 6": ("google_pixel_6-pictures-11037.php", "pixel-6"),
    "pixel 6a": ("google_pixel_6a-pictures-11229.php", "pixel-6a"),
    "pixel 6 pro": ("google_pixel_6_pro-pictures-10918.php", "pixel-6-pro"),
    "pixel 7": ("google_pixel_7-pictures-11903.php", "pixel-7"),
    "pixel 7a": ("google_pixel_7a-pictures-12170.php", "pixel-7a"),
    "pixel 8": ("google_pixel_8-pictures-12546.php", "pixel-8"),
    "pixel 8a": ("google_pixel_8a-pictures-12937.php", "pixel-8a"),
    "pixel 9 pro": ("google_pixel_9_pro-pictures-13218.php", "pixel-9-pro"),
    "pixel 9 pro fold": ("google_pixel_9_pro_fold-pictures-13220.php", "pixel-9-pro-fold"),
    "pixel 10 pro": ("google_pixel_10_pro_5g-pictures-13987.php", "pixel-10-pro"),
}


def valid_photo(url, prefix):
    p = urlsplit(url)
    if p.scheme != "https" or p.hostname not in (
        "fdn2.gsmarena.com", "fdn.gsmarena.com"
    ) or not p.path.startswith("/vv/pics/google/"):
        return False
    basename = p.path.rsplit("/", 1)[-1].lower()
    roots = ("google-" + prefix, "google-pixel" + prefix[len("pixel-"):])
    return any(re.fullmatch(re.escape(root) + r"-(?:1|2)\.jpg", basename)
               for root in roots)


def import_verified_pixels(apps, schema_editor):
    Device = apps.get_model("catalog", "Device")
    Image = apps.get_model("catalog", "DeviceImage")
    manifest = Path(__file__).resolve().parents[2] / "data" / "gsmarena_verified_pixel_images.csv"
    if not manifest.is_file():
        raise RuntimeError(f"Missing exact-model picture manifest: {manifest}")
    with manifest.open(encoding="utf-8-sig", newline="") as f:
        rows = list(csv.DictReader(f))
    if not rows:
        raise RuntimeError("GSMArena picture audit found no verified images")

    approved = []
    seen = set()
    shared = {}
    for row in rows:
        name = row["device"].casefold().strip()
        brand = row["brand"].casefold().strip()
        if brand != "google" or name not in PAGES or name in seen:
            raise RuntimeError(f"Unexpected/duplicated GSMArena model: {name}")
        seen.add(name)
        filename, prefix = PAGES[name]
        page = urlsplit(row["gallery_page"])
        if (page.scheme != "https" or page.hostname != "www.gsmarena.com"
                or page.path != "/" + filename or
                row["source_validation"] !=
                "GSMArena exact-model filename + JPEG response verified"):
            raise RuntimeError(f"Untrusted GSMArena source: {name}")
        try:
            urls = json.loads(row["image_urls"])
        except (TypeError, ValueError) as error:
            raise RuntimeError(f"Invalid photo JSON for {name}") from error
        if (not isinstance(urls, list) or not 1 <= len(urls) <= 2 or
            not all(isinstance(url, str) and valid_photo(url, prefix)
                    for url in urls) or len(set(urls)) != len(urls)):
            raise RuntimeError(f"Unsafe or mismatched image filename for {name}")
        for url in urls:
            if url in shared and shared[url] != name:
                raise RuntimeError(f"Image belongs to different phone: {url}")
            shared[url] = name
        approved.append((name, filename, urls))

    added = 0
    primary_changed = 0
    for name, filename, urls in approved:
        device = Device.objects.filter(is_published=True,
                                       brand__name__iexact="Google",
                                       name__iexact=name).first()
        if device is None:
            continue
        # GSMArena product shot is preferred over an interim illustration or
        # candid photo, while preserving every former genuine image as a
        # secondary gallery entry.
        Image.objects.filter(device_id=device.pk, is_primary=True).update(is_primary=False)
        for index, url in enumerate(urls):
            img, was_created = Image.objects.get_or_create(
                device_id=device.pk, image_url=url,
                defaults={
                    "view": "unspecified", "is_primary": index == 0,
                    "sort_order": index,
                    "alt_text": f"Google {device.name} GSMArena official product image",
                },
            )
            if not was_created and index == 0:
                img.is_primary = True
                img.sort_order = 0
                img.save(update_fields=["is_primary", "sort_order"])
            added += int(was_created)
        if device.primary_image_url != urls[0]:
            device.primary_image_url = urls[0]
            specs = dict(device.specifications or {})
            prior = specs.get("photo_attribution")
            if prior:
                specs["secondary_image_attribution"] = prior
            specs["photo_attribution"] = {
                "publisher": "GSMArena",
                "source": "https://www.gsmarena.com/" + filename,
                "reuse_basis": "GSMArena photo permission confirmed by Phone Discover operator",
                "verification": "exact-model CDN filename and actual JPEG bytes",
            }
            device.specifications = specs
            device.save(update_fields=["primary_image_url", "specifications", "updated_at"])
            primary_changed += 1
    print(f"GSMArena exact-model Pixel import: {len(approved)} checked model sources, "
          f"{added} new gallery views, {primary_changed} verified primary replacements. "
          "No catalog/price/variant changes.", flush=True)
