"""Bulk-add independently validated phone product photos across ten brands.

The manifest was generated by testing actual GSMArena JPEG responses for
exact catalog model filenames and rejecting shared/colliding pictures.
Operator has separately confirmed permission to use GSMArena photography.

This migration is transactionally applied by Django at normal Render deploy.
NO deletions, NO overwrites, NO new phone identities, NO price/variant changes.
A phone with existing primary/gallery photos is always left untouched.
"""
import csv
from collections import defaultdict
from pathlib import Path
from urllib.parse import urlsplit

from django.db import migrations

BRANDS = {
    "Apple": "apple", "Google": "google", "Infinix": "infinix",
    "Itel": "itel", "Motorola": "motorola", "Oppo": "oppo",
    "Samsung": "samsung", "Tecno": "tecno", "Vivo": "vivo",
    "Xiaomi": "xiaomi",
}
EVIDENCE = "Exact model filename; actual image/jpeg bytes verified"


def root_names(brand, name):
    import re
    import unicodedata
    text = unicodedata.normalize("NFKD", name).encode("ascii", "ignore").decode("ascii")
    text = re.sub(r"[^a-z0-9]+", "-", text.casefold().replace("+", " plus ")
                  .replace("&", " and ")).strip("-")
    prefix = BRANDS[brand]
    if text.startswith(prefix + "-"):
        text = text[len(prefix) + 1:]
    roots = [prefix + "-" + text]
    compact = re.sub(r"(?<=[a-z])-(?=[0-9])", "", text)
    if compact != text:
        roots.append(prefix + "-" + compact)
    return list(dict.fromkeys(roots))


def valid_photo(url, brand, name):
    parts = urlsplit(url)
    if (parts.scheme != "https" or parts.hostname != "fdn2.gsmarena.com"
        or parts.query or parts.fragment or not parts.path.lower().endswith(".jpg")):
        return False
    prefix = BRANDS[brand]
    for root in root_names(brand, name):
        for suffix in ("-1", "-00", "-0", "-2"):
            if parts.path == f"/vv/pics/{prefix}/{root}{suffix}.jpg":
                return True
        if parts.path == f"/vv/bigpic/{root}.jpg":
            return True
    return False


def backfill_verified_phone_images(apps, schema_editor):
    Device = apps.get_model("catalog", "Device")
    Image = apps.get_model("catalog", "DeviceImage")
    path = Path(__file__).resolve().parents[2] / "data" / "verified_brand_photos.csv"
    if not path.is_file():
        raise RuntimeError(f"Verified photo manifest missing: {path}")
    verified = []
    by_image = defaultdict(set)
    unique_models = set()
    with path.open(encoding="utf-8-sig", newline="") as source:
        reader = csv.DictReader(source)
        required = ("brand", "device", "image_url", "publisher",
                    "source_url", "source_validation")
        if not set(required).issubset(reader.fieldnames or []):
            raise RuntimeError("Incomplete verified photography manifest")
        for line, row in enumerate(reader, 2):
            brand = row["brand"].strip()
            name = row["device"].strip()
            key = (brand.casefold(), name.casefold())
            url = row["image_url"].strip()
            if (not name or brand not in BRANDS or key in unique_models or
                row["publisher"] != "GSMArena" or
                row["source_validation"] != EVIDENCE or
                row["source_url"] != url or
                not valid_photo(url, brand, name)):
                raise RuntimeError(f"Untrusted image source at line {line}")
            unique_models.add(key)
            by_image[url].add(key)
            verified.append((key, brand, name, url))
    if not verified:
        raise RuntimeError("No validated brand photo evidence to import")
    collisions = {url for url, owners in by_image.items() if len(owners) > 1}
    if collisions:
        raise RuntimeError("A single product image is claimed by two distinct phones")

    published = {
        (device.brand.name.casefold(), device.name.casefold()): device
        for device in Device.objects.select_related("brand").filter(is_published=True)
    }
    occupied = set(Image.objects.values_list("device_id", flat=True))
    creates = []
    changes = []
    ignored = 0
    for key, brand, name, url in verified:
        phone = published.get(key)
        if phone is None or phone.pk in occupied or phone.primary_image_url:
            ignored += 1
            continue
        creates.append(Image(
            device_id=phone.pk, image_url=url,
            view="unspecified", is_primary=True, sort_order=0,
            alt_text=f"{brand} {name} verified GSMArena product photo"[:200],
        ))
        specs = dict(phone.specifications or {})
        specs["photo_attribution"] = {
            "publisher": "GSMArena",
            "source": url,
            "reuse_basis": "permission confirmed by Phone Discover operator",
            "verification": EVIDENCE,
            "note": "Brand/model matched exactly; photo URL returned actual JPEG data.",
        }
        phone.primary_image_url = url
        phone.specifications = specs
        changes.append(phone)
        occupied.add(phone.pk)
    Image.objects.bulk_create(creates, batch_size=150)
    Device.objects.bulk_update(changes, ["primary_image_url", "specifications"],
                               batch_size=150)
    print(f"Verified multi-brand media: {len(verified)} exact source records; "
          f"{len(creates)} new phone images; {ignored} already pictured/unmatched. "
          "No prices, variants or existing media changed.", flush=True)


class Migration(migrations.Migration):
    dependencies = [("catalog", "0024_fill_pixel_5a_and_5a_5g_official_images")]
    operations = [migrations.RunPython(
        backfill_verified_phone_images, migrations.RunPython.noop
    )]
